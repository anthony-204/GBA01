        % Function to check type and convert values
function output = convertValue(~, val)
    if iscell(val) % If it's a cell array
        val = val{1}; % Extract the first element
    end

    % Handle different data types
    if isnumeric(val)
        output = val; % If numeric, leave it as is
    elseif ischar(val) || isstring(val)
        output = str2double(val); % Convert char or string to numeric if possible
        if isnan(output)
            output = string(val); % If conversion to numeric fails, convert to string
        end
    else
        output = val; % If some other type, leave it unchanged (or handle as needed)
    end
          end
    end

function output = convertValue(~, val)

 if iscell(val) % If it's a cell array

   val = val{1}; % Extract the first element

 end



 % Handle different data types

 if isnumeric(val)

   output = val; % If numeric, leave it as is

 elseif ischar(val) || isstring(val)

   output = str2double(val); % Convert char or string to numeric if possible

   if isnan(output)

       output = string(val); % If conversion to numeric fails, convert to string

   end

 else
 output = val; % If some other type, leave it unchanged (or handle as needed)

end




 end

        %% Function to check type and convert values
function output = convertValue(~, val)
    if iscell(val) % If it's a cell array
        val = val{1}; % Extract the first element
    end

      % Handle different data types
      if isnumeric(val)
          output = val; % If numeric, leave it as is
      elseif ischar(val) || isstring(val)
          output = str2double(val); % Convert char or string to numeric if possible
          if isnan(output)
              output = string(val); % If conversion to numeric fails, convert to string
          end
      else
          output = val; % If some other type, leave it unchanged (or handle as needed)
      end




            end
      end
